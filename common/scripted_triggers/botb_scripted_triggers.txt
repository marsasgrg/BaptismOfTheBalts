does_this_player_care_about_baltic = {
    this != root
    OR = {
		struggle:baltic_struggle = { is_culture_involved_in_struggle = prev.culture }
		struggle:baltic_struggle = { is_faith_involved_in_struggle = prev.faith }
	}
}

held_or_adherent_to_faith = {
	OR = {
		faith = $FAITH$
		holder = {
			faith = $FAITH$
		}
	}
}
held_or_adherent_to_culture = {
	OR = {
		culture = $CULTURE$
		holder = {
			culture = $CULTURE$
		}
	}
}
