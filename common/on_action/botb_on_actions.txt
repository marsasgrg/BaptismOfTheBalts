dreams_of_the_wolf = {
    trigger = {
        involved_activity ?= { has_activity_type = activity_hunt }
        culture = culture:lithuanian
        title:c_kernave.holder = this
        title:b_vilnius.title_province = {
            has_holding = no
        }
        religion = religion:baltic_religion
        highest_held_title_tier >= tier_duchy
        exists = cp:councillor_court_chaplain
        OR = {
            is_independent_ruler = yes
            highest_held_title_tier >= tier_kingdom
        }
        NOT = {
            OR = {
                has_character_flag = a_dream_of_the_wolf_begins
                has_character_flag = a_dream_of_the_wolf
                has_character_flag = rejected_wolf_dream
                has_variable = {
                    name = lizdeika
                }
            }
        }
    }
    effect = {
        
        random = {
            chance = 10
            trigger_event = {
                id = vilnius.0001
                days = 1
            } 
        }
    
    
    }
}

random_yearly_playable_pulse = {
    on_actions = {
        botb_random_yearly_playable_pulse
    }
}


botb_random_yearly_playable_pulse = {
    trigger = {
		basic_is_valid_for_yearly_events_trigger = yes
	}
    random_on_action = {
		chance_of_no_event = {
			value = 20
			if = {
				limit = {
					is_ai = yes
					highest_held_title_tier >= tier_kingdom
				}
				add = 30 # No need to waste performance here
			}
			else_if = {
				limit = {
					is_ai = yes
					highest_held_title_tier >= tier_duchy
				}
				add = 70 # No need to waste performance here
			}
			else_if = {
				limit = {
					is_ai = yes
				}
				add = 95 # AI counts aren't particularly important
			}
		}
        10 = baltic_flavor_yearly
	}
}

on_county_faith_change  = {
    on_actions = {botb_on_convert_county}
}

botb_on_convert_county = {
    effect = {
        if = {
            limit = {
                NOT = {
                    rite = {
                        rite_has_doctrine = unreformed_faith_doctrine
                    }
                }
                title_province = {
                    geographical_region = baltic_struggle_region     
                }
                NOR = {
                    exists = struggle:baltic_struggle
                    has_global_variable = baltic_struggle_finished
                }
 
            }
            start_struggle = {
                struggle_type = baltic_struggle
                start_phase = struggle_baltic_baptism 
            }
            every_county_in_region = {
                region = baltic_struggle_region
                faith = {
                    struggle:baltic_struggle = {
                        set_faith_as_involved = prev
                    }
                }
                culture = {
                    struggle:baltic_struggle = {
                        set_culture_as_involved = prev
                    }
                }
            }
        }
    }
}

    effect = {
        if = {

            }
        }
    }
