dreams_of_the_wolf = {
    trigger = {
        involved_activity ?= { has_activity_type = activity_hunt }
        culture = culture:lithuanian
        title:c_kernave.holder = this
        title:b_vilnius.title_province = {
            has_holding = no
        }
        religion = religion:baltic_religion
        highest_held_title_tier >= tier_duchy
        exists = cp:councillor_court_chaplain
        OR = {
            is_independent_ruler = yes
            highest_held_title_tier >= tier_kingdom
        }
        NOT = {
            OR = {
                has_character_flag = a_dream_of_the_wolf_begins
                has_character_flag = a_dream_of_the_wolf
                has_character_flag = rejected_wolf_dream

            }
        }
    }
    effect = {
        
        random = {
            chance = 10
            trigger_event = {
                id = vilnius.0001
                days = 1
            } 
        }
    
    
    }
}

random_yearly_playable_pulse = {
    on_actions = {
        botb_random_yearly_playable_pulse
    }
}


botb_random_yearly_playable_pulse = {
    trigger = {
		basic_is_valid_for_yearly_events_trigger = yes
	}
    random_on_action = {
		chance_of_no_event = {
			value = 20
			if = {
				limit = {
					is_ai = yes
					highest_held_title_tier >= tier_kingdom
				}
				add = 30 # No need to waste performance here
			}
			else_if = {
				limit = {
					is_ai = yes
					highest_held_title_tier >= tier_duchy
				}
				add = 70 # No need to waste performance here
			}
			else_if = {
				limit = {
					is_ai = yes
				}
				add = 95 # AI counts aren't particularly important
			}
		}
        10 = baltic_flavor_yearly
	}
}

baltic_flavor_yearly = {
    trigger = {
        botb_eligible_for_yearly_events = yes
    }
    random_events = {
        chance_to_happen = 100

        100 = baltic_flavor.0001
    }
}
