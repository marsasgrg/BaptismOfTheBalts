create_vilnius = {
    picture = {
		reference = "gfx/interface/illustrations/event_scenes/market_tribal.dds"
	}
    decision_group_type = major
    ai_goal = yes
    sort_order = 1000

    is_shown = {
        has_character_flag = a_dream_of_the_wolf
        title:c_kernave.holder = this
        title:b_vilnius.title_province = {
            has_holding = no
        }
        OR = {
            is_independent_ruler = yes
            highest_held_title_tier >= tier_kingdom
        }
    }
    is_valid = {
        has_title = title:c_kernave
        title:b_vilnius.title_province = {
            has_holding = no
        }
    }
    cost = {
        gold = 400
        prestige = 1000
    }
    ai_potential = {always = yes}
    ai_will_do = {
        base = 100
    }

    effect = {
        remove_character_flag = a_dream_of_the_wolf

        title:b_vilnius.title_province = {
            set_holding_type = tribal_holding
            add_building = tribe_02
        }
        title:b_vilnius = {
            set_capital_barony = yes
        }
        set_realm_capital = title:b_vilnius
        custom_tooltip = create_vilnius_bonus_choice
        trigger_event = vilnius.0003
    }

}

subsume_ruthenian_duchy = {
    decision_group_type = major
    ai_goal = yes
    sort_order = 10

    picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}

    is_shown = {
        OR = {
            AND = {
                this.culture = {
                    has_cultural_pillar = heritage_baltic
                }
                has_title = title:k_lithuania 
            }
            has_primary_title = title:k_lithuania
        }
        OR = {
            title:k_white_rus = {
                is_titular = no
            }
            title:k_ruthenia = {
                is_titular = no
            }
        }
    }
    is_valid = {
        
        OR = {
            AND = {
                title:k_white_rus = {
                    custom_tooltip = {
                        text = border_belarus_duchy_is_controlled
                        any_direct_de_jure_vassal_title = {
                            root = {
                                completely_controls = PREV
                            }
                            any_de_jure_county = {
                                any_title_to_title_neighboring_and_across_water_county = {
                                    kingdom = title:k_lithuania
                                }
                            }
                        }
                    }
                }
                OR = {
                    custom_tooltip = {
                        text = belarus_is_unheld
                    
                        NOT = {
                            exists = title:k_white_rus.holder
                        }
                    }
                    has_title = title:k_white_rus
                }
            }
            AND = {
                title:k_ruthenia = {
                    custom_tooltip = {
                        text = border_ukraine_duchy_is_controlled
                        any_direct_de_jure_vassal_title = {
                            root = {
                                completely_controls = PREV
                            }
                            any_de_jure_county = {
                                any_title_to_title_neighboring_and_across_water_county = {
                                    kingdom = title:k_lithuania
                                }
                            }
                        }
                    }
                }
                OR = {
                    custom_tooltip = {
                        text = ukraine_is_unheld
                    
                        NOT = {
                            exists = title:k_ruthenia.holder
                        }
                    }
                    has_title = title:k_ruthenia
                }
            }

        }
    }
    cooldown = { years = 15 }
    effect = {
        
        if = {
            limit = {                    
                OR = {
                    NOT = {
                        exists = title:k_ruthenia.holder
                    }
                    has_title = title:k_ruthenia
                }
            }
            title:k_ruthenia = {
                every_direct_de_jure_vassal_title = {
                    if = {
                        limit = {
                            root = {
                                completely_controls = PREV
                            }
                            any_de_jure_county = {
                                any_title_to_title_neighboring_and_across_water_county = {
                                    kingdom = title:k_lithuania
                                }
                            }
                        }
                        add_to_list = duchies_that_will_be_integrated
                    }
                }
            } 
        }

        if = {
            limit = {                    
                OR = {
                    NOT = {
                        exists = title:k_white_rus.holder
                    }
                    has_title = title:k_white_rus
                }
            }
            title:k_white_rus = {
                every_direct_de_jure_vassal_title = {
                    if = {
                        limit = {
                            root = {
                                completely_controls = PREV
                            }
                            any_de_jure_county = {
                                any_title_to_title_neighboring_and_across_water_county = {
                                    kingdom = title:k_lithuania
                                }
                            }
                        }
                        add_to_list = duchies_that_will_be_integrated
                    }
                }
            } 
        }

        every_in_list = {
            list = duchies_that_will_be_integrated
            set_de_jure_liege_title = title:k_lithuania
        }
    }
}
