apply_palemonic_lineage_to_kids = {
    every_child = {
        even_if_dead = yes
        add_trait = palemonid
        if = {
            limit = {
                is_female = no
            }
            apply_palemonic_lineage_to_kids = yes
        }
    }
}

#MAKES INCOMPATIBLE WITH MODS THAT ADD TRAITS REMOVED FROM BASTARDS
remove_inherited_descendent_traits_effect = {
	if = {
		limit = {
			has_trait = sayyid
			exists = scope:real_father
			NOT = {
				scope:real_father = { has_trait = sayyid }
			}
		}
		remove_trait = sayyid
	}
	if = {
		limit = {
			has_trait = golden_lineage
			exists = scope:real_father
			NOT = {
				scope:real_father = { has_trait = golden_lineage }
			}
		}
		remove_trait = golden_lineage
	}
    if = {
		limit = {
			has_trait = palemonid
			exists = scope:real_father
			NOT = {
				scope:real_father = { has_trait = palemonid }
			}
		}
		remove_trait = palemonid
	}
}

baltic_struggle_ending_culture_list_effect = {
	every_culture_global = {		
		limit = {
			save_temporary_scope_as = this_culture
			struggle:baltic_struggle = { is_culture_involved_in_struggle = scope:this_culture }
			culture_number_of_counties > 0
		}
		save_scope_as = culture_scope
		add_to_global_variable_list = {
			name = baltic_struggle_ending_culture_list
			target = scope:culture_scope
		}
	}
}

# Add all Involved independent rulers to a list
baltic_struggle_ending_involved_list_effect = {
	save_scope_as = struggle_ender
	struggle:baltic_struggle = {
		every_involved_ruler = {
			limit = {
				top_liege = this
				primary_title = { is_mercenary_company = no }
			}
			add_to_list = struggle_involvees
		}
	}
	# Add all involved cultures with counties to a list, for war checks
	baltic_struggle_ending_culture_list_effect = yes
	# Add all involved faiths with counties to a list, for marriage and holy war checks
	baltic_struggle_ending_faith_list_effect = yes
}

			}
			save_scope_as = faith_scope
			add_to_global_variable_list = {
				name = baltic_struggle_ending_faith_list
				target = scope:faith_scope
			}
		}
	}
}
